services:
  backend:
    build: ./backend
    container_name: modish_backend
    restart: always
    ports:
      - "8000:8000"
    env_file:
      - ./backend/.env
    depends_on:
      - postgres
      - redis

  postgres:
    image: postgres:16
    container_name: modish_postgres
    restart: always
    environment:
      POSTGRES_DB: modish
      POSTGRES_USER: modish_user
      POSTGRES_PASSWORD: modish_password_change_me
    volumes:
      - ./postgres_data:/var/lib/postgresql/data

  redis:
    image: redis:7
    container_name: modish_redis
    restart: always
    volumes:
      - ./redis_data:/data
